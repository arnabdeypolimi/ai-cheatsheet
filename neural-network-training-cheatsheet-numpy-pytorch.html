def adamw_step(w, g, m, v, t, lr=1e-3, b1=0.9, b2=0.999,
               eps=1e-8, wd=0.01):            # NumPy, t starts at 1
    m = b1 * m + (1 - b1) * g                 # 1st moment (mean)
    v = b2 * v + (1 - b2) * g ** 2            # 2nd moment (variance)
    m_hat, v_hat = m / (1 - b1 ** t), v / (1 - b2 ** t)  # bias fix
    w = w - lr * (m_hat / (np.sqrt(v_hat) + eps) + wd * w)
    return w, m, v

MLP from scratch (NumPy)

Backprop is the chain rule for a dense layer Z = XW + b: dW = Xᵀ·dZ, db = Σ dZ, dX = dZ·Wᵀ.

import numpy as np
def softmax(z):
    e = np.exp(z - z.max(1, keepdims=True))     # stable
    return e / e.sum(1, keepdims=True)

rng = np.random.default_rng(0)                  # X: [N, D], y: [N]
W1 = rng.normal(0, np.sqrt(2 / D), (D, H)); b1 = np.zeros(H)
W2 = rng.normal(0, np.sqrt(2 / H), (H, K)); b2 = np.zeros(K)
for epoch in range(200):
    Z1 = X @ W1 + b1; A1 = np.maximum(0, Z1)    # ReLU
    P = softmax(A1 @ W2 + b2)
    loss = -np.log(P[np.arange(N), y]).mean()
    dZ2 = P.copy(); dZ2[np.arange(N), y] -= 1   # dL/dz
    dZ2 /= N                                    # softmax-onehot
    dW2, db2 = A1.T @ dZ2, dZ2.sum(0)
    dZ1 = (dZ2 @ W2.T) * (Z1 > 0)               # ReLU'
    dW1, db1 = X.T @ dZ1, dZ1.sum(0)
    for p, g in [(W1, dW1), (b1, db1), (W2, dW2), (b2, db2)]:
        p -= lr * g                             # SGD step

model = nn.Sequential(nn.Linear(D, H), nn.ReLU(),
                      nn.Dropout(0.1), nn.Linear(H, K)).to(dev)
loss_fn = nn.CrossEntropyLoss()
opt = torch.optim.AdamW(model.parameters(), lr=1e-3)
sched = torch.optim.lr_scheduler.OneCycleLR(
    opt, max_lr=1e-3, total_steps=E * len(train_dl))
for epoch in range(E):
    model.train()                       # dropout / BN on
    for xb, yb in train_dl:
        xb, yb = xb.to(dev), yb.to(dev)
        loss = loss_fn(model(xb), yb)   # logits in
        opt.zero_grad()                 # grads accumulate!
        loss.backward()                 # autograd = backprop
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        opt.step(); sched.step()
    model.eval(); correct = 0
    with torch.no_grad():
        for xb, yb in val_dl:
            pred = model(xb.to(dev)).argmax(1)
            correct += (pred == yb.to(dev)).sum().item()
    acc = correct / len(val_dl.dataset)
